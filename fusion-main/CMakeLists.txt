cmake_minimum_required(VERSION 3.20)
project(Lightrod LANGUAGES C CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Lightrod requires Windows x64, matching Lunar's JVM.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
file(GLOB_RECURSE FUSION_SOURCES CONFIGURE_DEPENDS fusion/src/*.cpp)
file(GLOB IMGUI_SOURCES fusion/ext/imgui/*.cpp)
list(FILTER IMGUI_SOURCES EXCLUDE REGEX "imgui_demo.cpp$")
add_library(lightrod_native OBJECT ${FUSION_SOURCES} ${IMGUI_SOURCES}
    fusion/ext/minhook/buffer.c fusion/ext/minhook/hook.c
    fusion/ext/minhook/trampoline.c fusion/ext/minhook/hde/hde64.c
    fusion/src/base/patcher/miniz.c)
target_compile_definitions(lightrod_native PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS)
add_library(Lightrod SHARED $<TARGET_OBJECTS:lightrod_native>)
set(NATIVE_LIBRARIES "${CMAKE_CURRENT_SOURCE_DIR}/fusion/ext/jni/jvm.lib"
    opengl32 user32 gdi32 shell32 ole32 dwmapi imm32 uuid winmm)
target_link_libraries(Lightrod PRIVATE ${NATIVE_LIBRARIES})
set_target_properties(Lightrod PROPERTIES PREFIX "" OUTPUT_NAME "Lightrod")
if(MINGW)
    target_link_options(Lightrod PRIVATE -static -static-libgcc -static-libstdc++)
    target_compile_options(lightrod_native PRIVATE -fpermissive)
endif()
enable_testing()
add_executable(lightrod_tests tests/lightrod_tests.cpp
    fusion/src/base/moduleManager/modules/lightrod/config.cpp)
target_compile_definitions(lightrod_tests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(lightrod_tests PRIVATE user32)
if(MINGW)
    target_link_options(lightrod_tests PRIVATE -static -static-libgcc -static-libstdc++)
endif()
add_test(NAME Lightrod.behavior COMMAND lightrod_tests)
add_executable(lightrod_menu_smoke tests/menu_smoke.cpp $<TARGET_OBJECTS:lightrod_native>)
target_compile_definitions(lightrod_menu_smoke PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(lightrod_menu_smoke PRIVATE ${NATIVE_LIBRARIES})
if(MINGW)
    target_link_options(lightrod_menu_smoke PRIVATE -static -static-libgcc -static-libstdc++)
endif()
# The native objects import jvm.dll, which is present inside Lunar's process.
# Run this optional render harness with a compatible jvm.dll on PATH.

add_library(lightrod_loader STATIC injector/loader.cpp)
target_compile_definitions(lightrod_loader PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
target_link_libraries(lightrod_loader PUBLIC user32)
add_executable(LightrodInjector WIN32 injector/main.cpp)
set_target_properties(LightrodInjector PROPERTIES OUTPUT_NAME "Lightrod-Injector")
target_link_libraries(LightrodInjector PRIVATE lightrod_loader gdi32 comctl32)
add_library(lightrod_loader_probe SHARED tests/loader_probe.cpp)
set_target_properties(lightrod_loader_probe PROPERTIES PREFIX "" OUTPUT_NAME "Lightrod.LoaderProbe")
add_executable(lightrod_loader_tests tests/loader_tests.cpp)
target_link_libraries(lightrod_loader_tests PRIVATE lightrod_loader)
add_dependencies(lightrod_loader_tests lightrod_loader_probe)
if(MINGW)
    target_link_options(LightrodInjector PRIVATE -municode -static -static-libgcc -static-libstdc++)
    target_link_options(lightrod_loader_tests PRIVATE -municode -static -static-libgcc -static-libstdc++)
    target_link_options(lightrod_loader_probe PRIVATE -static -static-libgcc)
endif()
add_test(NAME Lightrod.loader COMMAND lightrod_loader_tests $<TARGET_FILE:lightrod_loader_probe>)
add_test(NAME Lightrod.injector-ui COMMAND LightrodInjector --smoke-test)
